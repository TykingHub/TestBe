    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - mousePos
        MainFrame.Position = UDim2.new(framePos.X.Scale, framePos.X.Offset + delta.X, framePos.Y.Scale, framePos.Y.Offset + delta.Y)

    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false

local function CreateTab(tabName)
    local TabButton = Instance.new("TextButton")
    TabButton.Name = tabName .. "Button"
    TabButton.Size = UDim2.new(1, -10, 0, 35)
    TabButton.Position = UDim2.new(0, 5, 0, 0)
    TabButton.BackgroundColor3 = Color3.fromRGB(15, 25, 38)
    TabButton.BackgroundTransparency = 0.5
    TabButton.Text = tabName
    TabButton.TextColor3 = Color3.fromRGB(170, 210, 240)
    TabButton.TextSize = 13
    TabButton.Font = Enum.Font.GothamMedium
    TabButton.Parent = TabList

    local TabCorner = Instance.new("UICorner")
    TabCorner.CornerRadius = UDim.new(0, 8)
    TabCorner.Parent = TabButton

    local TabContent = Instance.new("ScrollingFrame")
    TabContent.Name = tabName .. "Content"
    TabContent.Size = UDim2.new(1, 0, 1, 0)
    TabContent.Position = UDim2.new(0, 0, 0, 0)
    TabContent.BackgroundTransparency = 1
    TabContent.Visible = false
    TabContent.ScrollBarThickness = 3
    TabContent.Parent = Container

    local ContentLayout = Instance.new("UIListLayout")
    ContentLayout.SortOrder = Enum.SortOrder.LayoutOrder
    ContentLayout.Padding = UDim.new(0, 8)
    ContentLayout.Parent = TabContent

    if FirstTab then
        TabContent.Visible = true
        TabButton.BackgroundColor3 = Color3.fromRGB(0, 100, 160)
        TabButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        FirstTab = false
    end

    TabButton.MouseButton1Click:Connect(function()
        for _, content in pairs(Container:GetChildren()) do
            if content:IsA("ScrollingFrame") then content.Visible = false end
        end
        for _, btn in pairs(TabList:GetChildren()) do
            if btn:IsA("TextButton") then
                btn.BackgroundColor3 = Color3.fromRGB(15, 25, 38)
                btn.TextColor3 = Color3.fromRGB(170, 210, 240)
            end
        end
        TabContent.Visible = true
        TabButton.BackgroundColor3 = Color3.fromRGB(0, 100, 160)
        TabButton.TextColor3 = Color3.fromRGB(255, 255, 255)
    end)

    return TabContent
end

local function buyItem(itemName)
    local items = {
        ["sword"] = {"Sword", "Iron Sword", "Espada"},
        ["armor"] = {"Armor", "Iron Armor", "Armadura"},
        ["pickaxe"] = {"Pickaxe", "Pico"},
        ["axe"] = {"Axe", "Hacha"},
        ["blocks"] = {"Blocks", "Bloques", "Wool"},
        ["bow"] = {"Bow", "Arco"},
        ["arrows"] = {"Arrows", "Flechas"}
    }
    local names = items[itemName] or {itemName}
    local found = false
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("ClickDetector") or v:IsA("ProximityPrompt") then
            for _, n in ipairs(names) do
                if v.Parent and string.find(string.lower(v.Parent.Name), string.lower(n)) then
                    found = true
                    fireclickdetector(v)
                    pcall(function() v:InputHoldBegin() end)
                    return true
                end
            end
        end
    end
    if not found and autoBuyEnabled then
        local notif = Instance.new("Frame")
        notif.Size = UDim2.new(0, 250, 0, 40)
        notif.Position = UDim2.new(0.5, -125, 0, -50)
        notif.BackgroundColor3 = Color3.fromRGB(180, 40, 40)
        notif.BorderSizePixel = 0
        notif.Parent = ScreenGui
        Instance.new("UICorner", notif).CornerRadius = UDim.new(0, 6)
        local txt = Instance.new("TextLabel")
        txt.Size = UDim2.new(1, 0, 1, 0)
        txt.BackgroundTransparency = 1
        txt.Text = "No material suficiente"
        txt.TextColor3 = Color3.fromRGB(255, 255, 255)
        txt.TextSize = 14
        txt.Font = Enum.Font.GothamBold
        txt.Parent = notif
        notif:TweenPosition(UDim2.new(0.5, -125, 0, 10), "Out", "Quad", 0.5)
        wait(2)
        notif:TweenPosition(UDim2.new(0.5, -125, 0, -50), "In", "Quad", 0.5)
        wait(0.5)
        notif:Destroy()
    end
    return false
end

    end
end)
